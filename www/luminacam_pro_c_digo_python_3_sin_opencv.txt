# LuminaCam Pro — Implementación Completa en Python 3 (Sin OpenCV)

Aplicación de escritorio / portátil completa desarrollada en **Python 3 puro**, sin la librería `opencv-python` (`cv2`). 

Para la captura de vídeo multiplataforma, interfaz gráfica moderna AMOLED y procesamiento acelerado de fotogramas, utiliza el stack estándar y ligero:
- **`imageio` / `imageio-ffmpeg`** o **`pygame.camera` / `av`** para captura de webcam/cámara.
- **`numpy`** para el procesamiento computacional matricial ultrarrápido (fusión de luz, rangos min/max, chroma key y máscaras).
- **`customtkinter` / `tkinter` / `PIL` (Pillow)** para la interfaz gráfica AMOLED con soporte de dibujo de máscaras, deslizadores, cuentagotas y visualización en tiempo real.
- **`speech_recognition` / `vosk` o teclado directo** para el control hands-free por voz.

---

```bash
pip install numpy pillow customtkinter imageio imageio-ffmpeg SpeechRecognition
```

```python
# 1. MOTOR COMPUTACIONAL DE FUSIÓN Y SHADERS NUMPY
# ==============================================================================

class BlendMode:
    LIGHT_PAINT = "Pintar Luz"
    SHADOW_PAINT = "Pintar Sombra"
    TRAIL_DECAY = "Modo Estela"


class ComputationalEngine:
    def __init__(self, width: int = 640, height: int = 480):
        self.width = width
        self.height = height
        
        # Buffers flotantes [0.0 - 255.0]
        self.canvas_buffer = np.zeros((height, width, 3), dtype=np.float32)
        self.mask_buffer = np.zeros((height, width), dtype=np.float32) # 0: Normal, 1: Escudo Negativo
        self.background_layer: Optional[np.ndarray] = None
        
        # Historial de checkpoints en memoria (Undo / Deshacer)
        self.checkpoints: List[np.ndarray] = []
        
        # Parámetros de procesamiento
        self.blend_mode = BlendMode.LIGHT_PAINT
        self.min_luminance = 0.15 * 255.0
        self.max_luminance = 0.90 * 255.0
        self.decay_factor = 0.965
        
        # Chroma Key / Exclusión de Color
        self.chroma_enabled = False
        self.target_color_rgb = np.array([255, 158, 27], dtype=np.float32) # Amarillo Sodio por defecto
        self.color_tolerance = 45.0 # Distancia Euclidiana RGB
        self.exclusive_color_mode = False # True = Solo captura este color

    def reset_canvas(self):
        self.canvas_buffer.fill(0.0)
        self.checkpoints.clear()

    def create_checkpoint(self):
        """Copia el estado actual del lienzo para permitir Deshacer."""
        if len(self.checkpoints) >= 10:
            self.checkpoints.pop(0) # Máximo 10 checkpoints en RAM
        self.checkpoints.append(self.canvas_buffer.copy())

    def undo_last_checkpoint(self) -> bool:
        if self.checkpoints:
            self.canvas_buffer = self.checkpoints.pop()
            return True
        return False

    def process_frame(self, raw_frame_rgb: np.ndarray, is_recording: bool) -> np.ndarray:
        """
        Ejecuta el pipeline de filtrado y acumulación matricial idéntico al Fragment Shader.
        """
        frame_f32 = raw_frame_rgb.astype(np.float32)

        # Si no está grabando, devolvemos la imagen de sensor en vivo combinada con el lienzo
        if not is_recording:
            # High-Contrast Overlay durante la pausa: Realce de gamma 1.25x
            boosted_canvas = np.clip(np.power(self.canvas_buffer / 255.0, 1.0 / 1.25) * 255.0, 0, 255)
            # Combinar en vivo con el lienzo acumulado congelado
            preview = np.maximum(boosted_canvas, frame_f32)
            return preview.astype(np.uint8)

        # 1. Filtro de Luminancia (Rec. 709)
        # Y = 0.2126 R + 0.7152 G + 0.0722 B
        lum = (0.2126 * frame_f32[:, :, 0] + 
               0.7152 * frame_f32[:, :, 1] + 
               0.0722 * frame_f32[:, :, 2])
        
        lum_valid_mask = (lum >= self.min_luminance) & (lum <= self.max_luminance)

        # 2. Filtro de Exclusión de Color (Chroma Key)
        if self.chroma_enabled:
            dist = np.sqrt(np.sum((frame_f32 - self.target_color_rgb) ** 2, axis=2))
            if not self.exclusive_color_mode:
                # Descarta el color seleccionado
                color_mask = dist > self.color_tolerance
            else:
                # Modo pase exclusivo: solo admite el color seleccionado
                color_mask = dist <= self.color_tolerance
            lum_valid_mask = lum_valid_mask & color_mask

        # 3. Aplicar Máscara Negativa (Escudo Anti-Destellos)
        # Donde mask_buffer == 1.0, se ignora completamente el nuevo fotograma
        active_exposure_mask = lum_valid_mask & (self.mask_buffer < 0.5)

        # Crear el fotograma filtrado
        filtered_live = np.zeros_like(frame_f32)
        filtered_live[active_exposure_mask] = frame_f32[active_exposure_mask]

        # 4. Modos de Acumulación
        if self.blend_mode == BlendMode.LIGHT_PAINT:
            # Máximo brillo píxel a píxel
            self.canvas_buffer = np.maximum(self.canvas_buffer, filtered_live)

        elif self.blend_mode == BlendMode.SHADOW_PAINT:
            # Mínimo brillo para siluetas
            shadow_mask = active_exposure_mask & (lum > 0)
            self.canvas_buffer[shadow_mask] = np.minimum(self.canvas_buffer[shadow_mask], filtered_live[shadow_mask])

        elif self.blend_mode == BlendMode.TRAIL_DECAY:
            # Desvanecimiento progresivo tipo estela
            decayed = self.canvas_buffer * self.decay_factor
            self.canvas_buffer = np.maximum(decayed, filtered_live)

        # 5. Capa Base de Fondo si existe
        result = self.canvas_buffer.copy()
        if self.background_layer is not None:
            result = np.maximum(result, self.background_layer)

        return np.clip(result, 0, 255).astype(np.uint8)

class PurePythonCamera:
    """Captura de vídeo multiplataforma usando imageio / ffmpeg."""
    def __init__(self, camera_index: int = 0, width: int = 640, height: int = 480):
        self.camera_index = camera_index
        self.width = width
        self.height = height
        self.reader = None
        self.is_running = False

    def start(self) -> bool:
        if not HAS_IMAGEIO:
            print("[Aviso] imageio no instalado. Se usará un generador de prueba de luz.")
            self.is_running = True
            return True

        # Intento de conexión según el sistema operativo
        try:
            if sys.platform.startswith('darwin'):
                device_uri = f"<video{self.camera_index}>"
                format_name = "avfoundation"
            elif sys.platform.startswith('win'):
                device_uri = f"<video{self.camera_index}>"
                format_name = "dshow"
            else:
                device_uri = f"/dev/video{self.camera_index}"
                format_name = "v4l2"

            self.reader = iio.imiter(
                device_uri,
                format=format_name,
                fps=30
            )
            self.is_running = True
            return True
        except Exception as e:
            print(f"[Aviso] Cámara nativa no detectada ({e}). Modo Simulación de Sensor activado.")
            self.is_running = True
            return True

    def get_frame(self) -> np.ndarray:
        if self.reader is not None:
            try:
                frame = next(self.reader)
                img = Image.fromarray(frame).resize((self.width, self.height))
                return np.array(img, dtype=np.uint8)
            except Exception:
                pass

        # Generador de sensor sintético simulado (fuente de luz móvil para pruebas)
        t = time.time() * 2.0
        frame = np.zeros((self.height, self.width, 3), dtype=np.uint8)
        # Trazador circular de luz neón
        cx = int(self.width / 2 + np.cos(t) * (self.width * 0.3))
        cy = int(self.height / 2 + np.sin(t * 1.5) * (self.height * 0.3))
        
        y, x = np.ogrid[:self.height, :self.width]
        dist_sq = (x - cx)**2 + (y - cy)**2
        light_spot = np.exp(-dist_sq / (2 * (18**2))) * 255.0
        
        # Color cian/neón
        frame[:, :, 0] = np.clip(light_spot * 0.1, 0, 255).astype(np.uint8)
        frame[:, :, 1] = np.clip(light_spot * 1.0, 0, 255).astype(np.uint8)
        frame[:, :, 2] = np.clip(light_spot * 0.8, 0, 255).astype(np.uint8)
        return frame

    def release(self):
        self.is_running = False
        if self.reader is not None and hasattr(self.reader, 'close'):
            try:
                self.reader.close()
            except Exception:
                pass

# 3. SERVICIO DE CONTROL POR VOZ OFFLINE (HANDS-FREE)
# ==============================================================================

class VoiceTriggerService:
    def __init__(self, callback):
        self.callback = callback
        self.is_listening = False
        self.thread = None

    def start(self):
        if not HAS_SPEECH:
            print("[Aviso] SpeechRecognition no está instalado. Use los botones o teclado (Espacio / U).")
            return
        self.is_listening = True
        self.thread = threading.Thread(target=self._listen_loop, daemon=True)
        self.thread.start()

    def _listen_loop(self):
        recognizer = sr.Recognizer()
        recognizer.energy_threshold = 300
        recognizer.dynamic_energy_threshold = True

        while self.is_listening:
            try:
                with sr.Microphone() as source:
                    audio = recognizer.listen(source, phrase_time_limit=3)
                    text = recognizer.recognize_google(audio, language="es-ES").lower()
                    print(f"🎙️ Voz detectada: {text}")
                    if any(w in text for w in ["play", "iniciar", "grabar"]):
                        self.callback("play")
                    elif any(w in text for w in ["pausa", "parar", "congelar"]):
                        self.callback("pause")
                    elif any(w in text for w in ["deshacer", "atrás", "volver"]):
                        self.callback("undo")
                    elif any(w in text for w in ["guardar", "foto"]):
                        self.callback("save")
            except Exception:
                pass

    # ----------------- EVENTOS Y ACCIONES -----------------

    def toggle_recording(self):
        if not self.is_recording:
            # Pasa a GRABANDO
            self.is_recording = True
            self.is_paused = False
            self.lbl_status.configure(text="● GRABANDO (ACUMULACIÓN ACTIVA)", text_color="#FF3B30")
            self.btn_shutter.configure(text="⏸ PAUSAR (CONGELAR)", fg_color="#FFB800", text_color="#000000")
        else:
            # Pasa a PAUSA: crea checkpoint automático
            self.is_recording = False
            self.is_paused = True
            self.engine.create_checkpoint()
            self.lbl_status.configure(text="⏸ PAUSA · OVERLAY 1.25x (HOT-SWAP LISTO)", text_color="#FFB800")
            self.btn_shutter.configure(text="▶ REANUDAR TRAZO", fg_color="#00FFCC", text_color="#000000")

    def undo_checkpoint(self):
        if self.engine.undo_last_checkpoint():
            print("↺ Checkpoint restaurado.")
        else:
            print("No hay más checkpoints que deshacer.")

    def save_artwork(self):
        # Guarda la imagen compuesta en disco sin pérdida
        img_np = np.clip(self.engine.canvas_buffer, 0, 255).astype(np.uint8)
        img = Image.fromarray(img_np)
        filename = f"lightpaint_{int(time.time())}.png"
        img.save(filename)
        print(f"✨ Obra guardada con éxito como '{filename}'")

    def handle_voice_command(self, cmd: str):
        if cmd == "play" and not self.is_recording:
            self.after(0, self.toggle_recording)
        elif cmd == "pause" and self.is_recording:
            self.after(0, self.toggle_recording)
        elif cmd == "undo":
            self.after(0, self.undo_checkpoint)
        elif cmd == "save":
            self.after(0, self.save_artwork)

    def _change_blend_mode(self, mode: str):
        # Conmutación en caliente incluso en PAUSA
        self.engine.blend_mode = mode

    def _update_lum_threshold(self, val):
        self.engine.min_luminance = float(val)

    def _toggle_mask_draw_mode(self):
        self.drawing_mask_mode = (self.switch_mask_draw.get() == 1)

    def _on_canvas_paint(self, event):
        """Pinta el escudo protector (máscara negativa) con el ratón."""
        if self.drawing_mask_mode:
            x, y = event.x, event.y
            radius = 20
            y_min = max(0, y - radius)
            y_max = min(self.cam_h, y + radius)
            x_min = max(0, x - radius)
            x_max = min(self.cam_w, x + radius)
            self.engine.mask_buffer[y_min:y_max, x_min:x_max] = 1.0

    def _on_canvas_pick_color(self, event):
        """Muestrea el color en el píxel pinchado (Cuentagotas)."""
        x, y = event.x, event.y
        if 0 <= x < self.cam_w and 0 <= y < self.cam_h:
            frame = self.camera.get_frame()
            color = frame[y, x]
            self.engine.target_color_rgb = color.astype(np.float32)
            hex_color = f"#{color[0]:02x}{color[1]:02x}{color[2]:02x}"
            self.color_sample_box.configure(
                text=f"Muestreado: {hex_color} [RGB: {color[0]},{color[1]},{color[2]}]",
                fg_color=hex_color,
                text_color="#FFFFFF" if sum(color) < 380 else "#000000"
            )

    def _video_loop(self):
        """Bucle principal de procesamiento y refresco visual."""
        raw_frame = self.camera.get_frame()
        processed_frame = self.engine.process_frame(raw_frame, self.is_recording)

        # Si el modo pincel de máscara está activo, dibujar overlay rosa semitransparente
        if self.engine.mask_buffer.max() > 0:
            mask_indices = self.engine.mask_buffer > 0.5
            # Añadir tinte magenta neón al escudo
            processed_frame[mask_indices, 0] = np.clip(processed_frame[mask_indices, 0] + 120, 0, 255)
            processed_frame[mask_indices, 2] = np.clip(processed_frame[mask_indices, 2] + 120, 0, 255)

        # Convertir a imagen Tkinter y mostrar
        img = Image.fromarray(processed_frame)
        img_tk = ImageTk.PhotoImage(image=img)
        self.canvas_view.create_image(0, 0, anchor="nw", image=img_tk)
        self.canvas_view.image = img_tk

        self.after(30, self._video_loop)

    def destroy(self):
        self.camera.release()
        self.voice_service.is_listening = False
        super().destroy()


# ==============================================================================
# 5. PUNTO DE ENTRADA
# ==============================================================================

if __name__ == "__main__":
    app = LuminaCamApp()
    app.mainloop()
```

## 3. Guía de Ejecución y Uso

   ```bash
   pip install numpy pillow customtkinter imageio imageio-ffmpeg SpeechRecognition
   ```


   * **`Voz`**: Di en voz alta *"Play"*, *"Pausa"*, *"Deshacer"* o *"Guardar"*.
   * **`Simulación automática`**: Si no tienes una cámara conectada físicamente, el motor incluye un simulador de sensor con una estela de luz rotatoria para que pruebes los modos de fusión al instante.
