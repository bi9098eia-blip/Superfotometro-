## 3. Shader de Acumulación GPU (`app/src/main/res/raw/accumulate_frag.glsl`)
```glsl
#version 320 es
precision highp float;

in vec2 vTexCoord;
out vec4 fragColor;

uniform sampler2D uLiveTexture;
uniform sampler2D uAccumulatedTexture;
uniform sampler2D uMaskTexture;

uniform int uBlendMode;       // 0: Light, 1: Shadow, 2: Trail Decay
uniform float uDecayFactor;   // ej. 0.965
uniform float uMinLuminance;  // ej. 0.15
uniform float uMaxLuminance;  // ej. 0.90
uniform bool uIsPaused;
uniform float uGammaBoost;    // 1.25

float getLum(vec3 c) {
    return dot(c, vec3(0.2126, 0.7152, 0.0722));
}

void main() {
    vec4 live = texture(uLiveTexture, vTexCoord);
    vec4 prev = texture(uAccumulatedTexture, vTexCoord);
    vec4 mask = texture(uMaskTexture, vTexCoord);

    if (uIsPaused) {
        vec3 boosted = pow(prev.rgb, vec3(1.0 / uGammaBoost));
        fragColor = vec4(boosted, 1.0);
        return;
    }

    // Escudo Negativo de Protección (Canal G de la Máscara)
    if (mask.g > 0.1) {
        fragColor = prev;
        return;
    }

    float lum = getLum(live.rgb);
    if (lum < uMinLuminance || lum > uMaxLuminance) {
        live = vec4(0.0);
    }

    vec4 outColor = prev;
    if (uBlendMode == 0) {
        // Pintar Luz (Lighten / Max)
        outColor = max(prev, live);
    } else if (uBlendMode == 1) {
        // Pintar Sombra (Darken / Min)
        if (lum > 0.0) {
            outColor = min(prev, live);
        }
    } else if (uBlendMode == 2) {
        // Modo Estela (Decay)
        outColor = max(prev * uDecayFactor, live);
    }

    fragColor = outColor;
}
```

enum class CaptureState { READY, RECORDING, PAUSED }
enum class BlendMode(val title: String) { LIGHT("Pintar Luz"), SHADOW("Pintar Sombra"), TRAIL("Modo Estela") }

class MainActivity : ComponentActivity() {

    private val requestPermissions = registerForActivityResult(
        ActivityResultContracts.RequestMultiplePermissions()
    ) { _ -> }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        
        // Solicitar permisos en tiempo de ejecución
        val permissions = arrayOf(Manifest.permission.CAMERA, Manifest.permission.RECORD_AUDIO)
        if (permissions.any { ContextCompat.checkSelfPermission(this, it) != PackageManager.PERMISSION_GRANTED }) {
            requestPermissions.launch(permissions)
        }

        setContent {
            LuminaCamCloudApp()
        }
    }
}

@Composable
fun LuminaCamCloudApp() {
    var state by remember { mutableStateOf(CaptureState.READY) }
    var blendMode by remember { mutableStateOf(BlendMode.LIGHT) }

    Box(

            .background(AmoledBlack)

        // 1. Simulación / Render de Sensor de Visor

        }

            modifier = Modifier
                .fillMaxWidth()

        Row(

                .padding(bottom = 110.dp)

                .border(1.dp, Color.White.copy(alpha = 0.15f), RoundedCornerShape(24.dp))
                .padding(4.dp),
            horizontalArrangement = Arrangement.spacedBy(4.dp)

            BlendMode.values().forEach { mode ->
                val selected = blendMode == mode

                    modifier = Modifier
                        .clip(RoundedCornerShape(20.dp))

                        CaptureState.PAUSED -> CaptureState.RECORDING

            Icon(
                imageVector = when (state) {
                    CaptureState.RECORDING -> Icons.Default.Pause
                    CaptureState.PAUSED -> Icons.Default.PlayArrow
                    CaptureState.READY -> Icons.Default.PlayArrow

                contentDescription = null,
                tint = Color.Black,
                modifier = Modifier.size(36.dp)

## 5. Instrucciones para Ejecutar en la Nube (Google Cloud Workstations / Firebase Test Lab)

1. **Google Cloud Workstations (Android Studio in the Cloud)**:
   - Crea una estación de trabajo con la imagen base **Android**.
   - Clona tu repositorio o pega estos archivos en el directorio del proyecto.
   - Abre la terminal integrada de Cloud Workstations y ejecuta:
     ```bash
     ./gradlew assembleDebug
     ```
2. **Pruebas en Dispositivos Virtuales / Físicos en la Nube (Firebase Test Lab)**:
   - Para probar la app en cientos de teléfonos reales sin tener hardware local:
     ```bash
     gcloud firebase test android run \
       --type robo \
       --app app/build/outputs/apk/debug/app-debug.apk \
       --device model=oriole,version=33,locale=es,orientation=portrait
     ```
3. **Compilación sin Errores**:
   - Este código incluye soporte explícito para la arquitectura `x86_64` (emuladores de la nube) y `arm64-v8a` (dispositivos físicos).
